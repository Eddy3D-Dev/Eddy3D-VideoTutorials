namespace Shots

    /// <summary>A staged capability: real contract, canned answers. State changes only when the script says so.</summary>
    public sealed class Fake : ISetupCapability
    {
        public string Id { get; set; }
        public string Title { get; set; }
        public string Subtitle { get; set; }
        public IReadOnlyList<string> Routes { get; set; } = Array.Empty<string>();
        public SetupState State { get; set; } = SetupState.Missing;
        public string StateReason { get; set; } = string.Empty;
        public IReadOnlyList<SetupOption> Options { get; set; } = new List<SetupOption>();
        public IReadOnlyList<string> Diagnostics { get; set; } = new List<string>();
        public IReadOnlyList<string> Plugins { get; set; } = Array.Empty<string>();
        public SetupJob Job { get; set; }
        public Action OnRefresh;
        public Task RefreshAsync(CancellationToken ct)
        {
            OnRefresh?.Invoke();
            return Task.Delay(150, ct);
        }
    }

    public sealed class FakeCatalog : ISetupCatalog
    {
        public HostProfile Host { get; } = new HostProfile
        {
            OsDescription = "Windows 11 Education 10.0.26300",
            Architecture = "x64",
            ProcessorCount = 16,
            GpuDescription = "NVIDIA GeForce RTX 4070",
            ContainerDescription = "",
            WslDescription = "WSL 2",
            FreeDiskBytes = 412L * 1024 * 1024 * 1024,
            TotalMemoryBytes = 64L * 1024 * 1024 * 1024,
            AvailableMemoryBytes = 41L * 1024 * 1024 * 1024,
        };
        public List<ISetupCapability> List { get; } = new List<ISetupCapability>();
        public IReadOnlyList<ISetupCapability> Capabilities => List;
        public Task DetectHostAsync(CancellationToken ct) => Task.Delay(100, ct);
    }

    public static class Program
    {
        const long Total = (long)(1.5 * 1024 * 1024 * 1024);
        static string Out;
        static SemaphoreSlim Gate = new SemaphoreSlim(0);
        static double Target;      // 0..1, set by the script before releasing the gate
        static FakeCatalog Cat;
        static Fake Container, OpenFoam;
        static bool Pulled;
        static Form Win;

        [STAThread]
        public static int Main(string[] args)
        {
            Out = args.Length > 0 ? args[0] : Path.Combine(Path.GetTempPath(), "shots");
            Directory.CreateDirectory(Out);
            var app = new Application(Eto.Platform.Detect);

            Cat = new FakeCatalog();
            Container = new Fake
            {
                Id = "container", Title = "Container runtime (Podman · Docker)", Subtitle = "runs every solver image on this machine",
                Plugins = Array.Empty<string>(), State = SetupState.Missing,
                StateReason = "No container runtime found. Install Podman (https://podman-desktop.io/downloads) or Docker Desktop, start it, then press Re-check.",
                Diagnostics = new[] { "container cli: none found (EDDY3D_CONTAINER_CLI overrides)" },
                Options = new List<SetupOption> { new SetupOption("download", "Open the Podman download page",
                    "Opens https://podman-desktop.io/downloads in your browser. Install Podman Desktop (or Docker Desktop), start it, then press Re-check.",
                    (p, c) => Task.FromResult(SetupResult.WaitingForUser("Download page opened — install and start Podman, then press Re-check."))) },
            };
            OpenFoam = new Fake
            {
                Id = "openfoam", Title = "(U)RANS CFD (OpenFOAM)", Subtitle = "simpleFoam · pimpleFoam · buoyantFoam · multicomponentFluid",
                Plugins = new[] { SetupPlugins.Outdoor, SetupPlugins.Indoor }, State = SetupState.Missing,
            };
            OpenFoam.OnRefresh = () =>
            {
                if (Pulled)
                {
                    OpenFoam.State = SetupState.Ready; OpenFoam.StateReason = ""; OpenFoam.Options = new List<SetupOption>();
                    OpenFoam.Routes = new[] { "containerized" };
                    OpenFoam.Diagnostics = new[] { @"container cli: C:\Program Files\RedHat\Podman\podman.exe", "wind image: docker.io/dicehub/openfoam:12@sha256:d7bbd9e7105718e16971fa3918e3a43e03bca8ae54da00423ca90001d326760e (present)" };
                }
                else
                {
                    OpenFoam.State = SetupState.Missing;
                    OpenFoam.StateReason = "No OpenFOAM engine is usable: no wind image pulled, no blueCFD install, no OpenFOAM 12 in WSL.";
                    OpenFoam.Options = new List<SetupOption> { PullOption(), BlueOption() };
                    OpenFoam.Diagnostics = new[] { @"container cli: C:\Program Files\RedHat\Podman\podman.exe", "wind image: docker.io/dicehub/openfoam:12@sha256:d7bbd9e7105718e16971fa3918e3a43e03bca8ae54da00423ca90001d326760e (not pulled)" };
                }
            };
            Cat.List.Add(Container);
            Cat.List.Add(OpenFoam);
            Cat.List.Add(Plain("umf", "Fully-coupled multi-region CFD / HAM / RAD / VEG (urbanMicroclimateFoam)", "urbanMicroclimateFoam · image, blueCFD-Core 2024, or a WSL build", SetupPlugins.OutdoorPlus));
            Cat.List.Add(Plain("radep", "Comfort study (Radiance · EnergyPlus)", "Radiance · EnergyPlus 26.1.0", SetupPlugins.Comfort));
            Cat.List.Add(Plain("openlb", "LBM CFD (OpenLB)", "CPU · OpenLB image or native solver", SetupPlugins.Lbm));
            Cat.List.Add(Plain("fluidx3d", "LBM CFD (FluidX3D)", "GPU · FluidX3D source, built on first run", SetupPlugins.FluidX3D, SetupPlugins.Lbm));
            Cat.List.Add(Plain("wrf", "Mesoscale weather · Weather Research and Forecasting (WRF)", "WRF 4.6.1 · WPS 4.6.0", SetupPlugins.Wrf));
            Cat.List.Add(Plain("palm", "Urban microclimate · LES / RANS (PALM-4U)", "PALM 25.10.1 · radiation · surfaces · canopy · biometeorology", SetupPlugins.Palm));
            Container.OnRefresh = null;

            var view = new Eddy3DSetupView(Cat, ConsoleSetupHost.Instance);
            var (client, minimum) = Eddy3DSetupView.FitTo(new Size(1500, 960));
            Win = new Form { Title = "Eddy3D Setup", ClientSize = client, MinimumSize = minimum, Content = view, ShowInTaskbar = false, Location = new Point(-2600, 40) };
            Console.Error.WriteLine("client " + client + " min " + minimum);
            Win.Shown += (s, e) => Application.Instance.AsyncInvoke(async () =>
            {
                try { await Script(view); }
                catch (Exception ex) { File.WriteAllText(Path.Combine(Out, "ERROR.txt"), ex.ToString()); }
                File.WriteAllText(Path.Combine(Out, "DONE.txt"), "done");
                Application.Instance.Quit();
            });
            app.Run(Win);
            return 0;
        }

        static Fake Plain(string id, string title, string sub, params string[] plugins) => new Fake
        {
            Id = id, Title = title, Subtitle = sub, Plugins = plugins, State = SetupState.Missing,
            StateReason = "Not installed.",
        };

        static SetupOption BlueOption() => new SetupOption(
            "bluecfd", "Install blueCFD-Core 2024",
            "Downloads blueCFD-Core 2024 and installs it to C:\\blueCFD-Core\\2024 (a space-free path the OpenFOAM toolchain can build through), then grants standard users write access so the later solver build and core patch need no further prompt. Asks for administrator once.",
            (p, c) => Task.FromResult(SetupResult.Failure("x", "x")),
            estimate: "~5 min · 2 GB", estimatedBytes: 2L * 1024 * 1024 * 1024, requiresElevation: true);

        static SetupOption PullOption() => new SetupOption(
            "of-images", "Pull the OpenFOAM wind image",
            "Pulls the OpenFOAM 12 image the wind and indoor solvers run in, with podman.",
            Run, estimate: "~4 min · 1.5 GB", estimatedBytes: Total, requiresCapabilityId: "container");

        static async Task<SetupResult> Run(IProgress<SetupProgress> p, CancellationToken ct)
        {
            const string stage = "Pulling the OpenFOAM wind image";
            p.Report(new SetupProgress("Pulling docker.io/dicehub/openfoam:12 (" + InstallProgress.Bytes(Total) + ")…", stage, 0, 0, 0, InstallProgress.Bytes(0) + " of " + InstallProgress.Bytes(Total)));
            while (true)
            {
                await Gate.WaitAsync(ct);
                var f = Target;
                if (f >= 1.0) break;
                p.Report(new SetupProgress("", stage, f, 0, 0, InstallProgress.Bytes((long)(Total * f)) + " of " + InstallProgress.Bytes(Total)));
            }
            Pulled = true;
            return SetupResult.Success("OpenFOAM wind image pulled.");
        }

        static async Task Script(Eddy3DSetupView view)
        {
            await Task.Delay(120);
            await Shot("00a_detecting");
            await Task.Delay(3400);                       // first detection settles
            await Shot("00_no_runtime");

            // Podman Desktop has started the machine: the container row becomes Ready.
            Container.State = SetupState.Ready; Container.StateReason = "";
            Container.Routes = new[] { "podman", "8 of 16 cores" };
            Container.Options = new List<SetupOption>();
            Container.Diagnostics = new[] { "container cli: podman", "daemon: 8 cores of 16" };
            Cat.Host.ContainerDescription = "podman";
            OpenFoam.OnRefresh();
            view.BeginDetect();
            await Task.Delay(3000);
            await Shot("01_podman_ready");

            ClickButton("Outdoor");
            await Task.Delay(1200);
            await Shot("01b_outdoor_tab");
            ClickButton("Pull");
            await Task.Delay(1200);
            await Shot("02_pull_000");
            var t0 = DateTime.UtcNow;
            double PullSeconds = Environment.GetEnvironmentVariable("SHOTS_FAST") == "1" ? 6 : 220;
            for (double pct = 2; pct <= 98; pct += 2)
            {
                var due = t0.AddSeconds(pct / 100.0 * PullSeconds);
                var wait = due - DateTime.UtcNow;
                if (wait > TimeSpan.Zero) await Task.Delay(wait);
                Target = pct / 100.0; Gate.Release();
                await Task.Delay(700);
                await Shot("02_pull_" + ((int)pct).ToString("000"));
            }
            Target = 1.0; Gate.Release();
            await Task.Delay(1500);
            view.BeginDetect();
            await Task.Delay(2500);
            await Shot("03_ready");
        }

        static void ClickButton(string text)
        {
            var root = Root();
            foreach (var b in Descend(root).OfType<SWC.Button>())
                if (ButtonText(b) == text && b.IsVisible && b.IsEnabled)
                {
                    b.RaiseEvent(new SW.RoutedEventArgs(SWC.Primitives.ButtonBase.ClickEvent));
                    return;
                }
            throw new Exception("no enabled button '" + text + "'");
        }

        static string ButtonText(SWC.Button b)
        {
            if (b.Content is string cs) return cs.Trim();
            foreach (var d in Descend(b)) if (d is SWC.TextBlock t && !string.IsNullOrWhiteSpace(t.Text)) return t.Text.Trim();
            return "";
        }

        static SW.FrameworkElement Root() => (SW.FrameworkElement)((SW.Window)Win.ControlObject).Content;

        static IEnumerable<SW.DependencyObject> Descend(SW.DependencyObject o)
        {
            yield return o;
            int n = SWM.VisualTreeHelper.GetChildrenCount(o);
            for (int i = 0; i < n; i++)
                foreach (var d in Descend(SWM.VisualTreeHelper.GetChild(o, i))) yield return d;
        }

        static async Task Shot(string name)
        {
            await Task.Delay(250);
            var root = Root();
            double w = root.ActualWidth, h = root.ActualHeight;
            const double scale = 2.0;
            var rtb = new SWMI.RenderTargetBitmap((int)(w * scale), (int)(h * scale), 96 * scale, 96 * scale, SWM.PixelFormats.Pbgra32);
            rtb.Render(root);
            var enc = new SWMI.PngBitmapEncoder();
            enc.Frames.Add(SWMI.BitmapFrame.Create(rtb));
            using (var fs = File.Create(Path.Combine(Out, name + ".png"))) enc.Save(fs);

            var sb = new StringBuilder();
            sb.Append("{\"name\":\"" + name + "\",\"w\":" + w.ToString("0.#", System.Globalization.CultureInfo.InvariantCulture) + ",\"h\":" + h.ToString("0.#", System.Globalization.CultureInfo.InvariantCulture) + ",\"items\":[");
            bool first = true;
            foreach (var d in Descend(root))
            {
                if (!(d is SW.FrameworkElement fe) || !fe.IsVisible || fe.ActualWidth < 2) continue;
                string kind = null, text = null;
                if (fe is SWC.Button b) { kind = "button"; text = ButtonText(b); }
                else if (fe is SWC.ProgressBar) kind = "progress";
                else if (fe is SWC.TextBlock tb && !string.IsNullOrWhiteSpace(tb.Text)) { kind = "text"; text = tb.Text; }
                else if (fe is SWC.Primitives.ToggleButton) kind = "toggle";
                else if (fe is SWC.TabItem) kind = "tab";
                if (kind == null) continue;
                var r = fe.TransformToAncestor(root).TransformBounds(new SW.Rect(0, 0, fe.ActualWidth, fe.ActualHeight));
                if (!first) sb.Append(','); first = false;
                sb.Append("{\"k\":\"" + kind + "\",\"t\":" + Json(text) + ",\"x\":" + F(r.X) + ",\"y\":" + F(r.Y) + ",\"w\":" + F(r.Width) + ",\"h\":" + F(r.Height) + "}");
            }
            sb.Append("]}");
            File.WriteAllText(Path.Combine(Out, name + ".json"), sb.ToString());
        }

        static string F(double v) => v.ToString("0.#", System.Globalization.CultureInfo.InvariantCulture);
        static string Json(string s) => s == null ? "null" : "\"" + s.Replace("\\", "\\\\").Replace("\"", "\\\"").Replace("\r", " ").Replace("\n", " ") + "\"";
    }
}
